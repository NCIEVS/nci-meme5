#
# Sample cron entries for NCI-META operational scripts.
#
# The csh scripts self-bootstrap from the production setenv.sh when APP_DIR is
# missing, so these entries can match the direct server invocation style.
#

#
# Run nightly tasks
# 1. Daily editing report
# 2. Repartition
# 3. This includes restarting the tomcat server.
#
00 23 * * 0-6 /bin/tcsh -f /local/content/MEME/MEME5/ncim/bin/nightlyTasks.csh > /local/content/MEME/MEME5/ncim/logs/nightlyTasks.log 2>&1

#
# Run weekly tasks
# 1. Create deep relationships table
# 2. This includes restarting the tomcat server.
#
00 01 * * 7 /bin/tcsh -f /local/content/MEME/MEME5/ncim/bin/weeklyTasks.csh > /local/content/MEME/MEME5/ncim/logs/weeklyTasks.log 2>&1

#
# Run cron validation script (runs the MID Validation queries and emails them)
#
00 09 * * 6 /bin/tcsh -f /local/content/MEME/MEME5/ncim/bin/cronvalidate.csh > /local/content/MEME/MEME5/ncim/logs/cronvalidate.log 2>&1
