#
# Sample cron entries for NCI-META operational scripts.
#
# Common runtime values are exported by the production setenv.sh. The csh
# scripts inherit those values from the bash wrapper below.
#
SETENV=/local/content/MEME/MEME5/ncim/setenv.sh
MEME_BIN=/local/content/MEME/MEME5/ncim/bin

#
# Run nightly tasks:
# 1. Daily editing report (always)
# 2. MUTUALLY_EXCLUSIVE repartition (when project automations are enabled)
# 3. Optional application restart (when project automations are enabled and
#    RESTART_SERVER_AFTER_NIGHTLY is true)
#
00 00 * * 0-5 /bin/bash -lc 'source "$SETENV" && exec "$MEME_BIN"/nightlyTasks.csh'

#
# Run the MID Validation process.
#
00 09 * * 6 /bin/bash -lc 'source "$SETENV" && exec "$MEME_BIN"/cronvalidate.csh'
